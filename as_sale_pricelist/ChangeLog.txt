### Consultas SQL ejecutadas
```sql
-- Eliminar la restricción foreign key existente
ALTER TABLE sale_order DROP CONSTRAINT IF EXISTS sale_order_last_promo_id_fkey;

-- Actualizar todos los registros con last_promo_id inválido a NULL
UPDATE sale_order
SET last_promo_id = NULL
WHERE last_promo_id IS NOT NULL 
  AND NOT EXISTS (
    SELECT 1 FROM coupon_program WHERE id = sale_order.last_promo_id
  );

-- Agregar la nueva restricción de clave foránea
ALTER TABLE sale_order
ADD CONSTRAINT sale_order_last_promo_id_fkey
FOREIGN KEY (last_promo_id) REFERENCES coupon_program(id) ON DELETE SET NULL;

-- Eliminar tablas de wizard persistentes (ahora son transitorias)
DROP TABLE IF EXISTS sale_order_pricelist_wizard CASCADE;
DROP TABLE IF EXISTS sale_order_pricelist_wizard_line CASCADE;
```
